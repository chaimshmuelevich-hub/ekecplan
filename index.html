-- טבלת פרויקטים
create table projects (
  id uuid default gen_random_uuid() primary key,
  created_at timestamptz default now(),
  name text not null, type text, area integer,
  floors integer, points integer, rooms integer,
  amperage text, notes text, ai_plan text,
  total_cost integer, source text,
  status text default 'quote',
  client_email text, assigned_to text,
  start_date date, end_date date
);

-- טבלת הוצאות
create table expenses (
  id uuid default gen_random_uuid() primary key,
  created_at timestamptz default now(),
  description text, amount integer,
  category text, project_id uuid
);
